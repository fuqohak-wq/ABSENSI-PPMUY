function renderTabelRekap(data) { const tbody = document.querySelector("#tableRekap tbody"); tbody.innerHTML = ""; // Inisialisasi variabel total let totalTagihan = 0; let totalBayar = 0; let totalSisa = 0; data.forEach(item => { // Hitung akumulasi totalTagihan += item.totalFinal; totalBayar += item.sudahTerbayar; totalSisa += item.belumTerbayar; let tr = document.createElement("tr"); tr.innerHTML = ` ${item.jenisPembayaran} ${item.jumlahSantri} ${item.nominalPerSantri !== "-" ? "Rp " + Number(item.nominalPerSantri).toLocaleString('id-ID') : "-"} Rp ${Number(item.totalFinal).toLocaleString('id-ID')} Rp ${Number(item.sudahTerbayar).toLocaleString('id-ID')} Rp ${Number(item.belumTerbayar).toLocaleString('id-ID')} `; tbody.appendChild(tr); }); // Tambahkan Baris Total Paling Bawah let trTotal = document.createElement("tr"); trTotal.className = "total-row"; trTotal.innerHTML = ` TOTAL SEMUA Rp ${totalTagihan.toLocaleString('id-ID')} Rp ${totalBayar.toLocaleString('id-ID')} Rp ${totalSisa.toLocaleString('id-ID')} `; tbody.appendChild(trTotal); }
